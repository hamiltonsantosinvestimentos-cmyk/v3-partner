-- Restaura a leitura total de SDR e CLOSER em sdr_leads e sdr_conversas.
alter policy "sdr_leads_partner_or_staff" on public.sdr_leads
  using (
    (partner_id = auth.uid())
    or exists (
      select 1 from public.profiles
      where profiles.id = auth.uid()
        and profiles.role = any (array['ADMIN'::user_role, 'GESTAO'::user_role, 'SDR'::user_role, 'CLOSER'::user_role])
    )
  );
